# 数据库表结构设计

> 以下为生产环境建议的表结构（SQLite / PostgreSQL 通用，PostgreSQL 请将 `INTEGER PRIMARY KEY AUTOINCREMENT` 改为 `SERIAL`/`BIGSERIAL`，`DATETIME` 改为 `TIMESTAMP`）。
> Demo 使用 `localStorage` 实现等价逻辑，字段与下表一一对应。

## 实体关系

```
users 1──* word_mastery *──1 words
users 1──* check_ins
users 1──* user_milestones *──1 milestones
users 1──* points_log
users 1──* exam_records
```

## 1. users（用户）

```sql
CREATE TABLE users (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  nickname      TEXT NOT NULL DEFAULT '小宇航员',
  title         TEXT NOT NULL DEFAULT '单词萌新',   -- 当前称号
  points        INTEGER NOT NULL DEFAULT 0,          -- 累计积分
  streak        INTEGER NOT NULL DEFAULT 0,          -- 连续打卡天数
  last_check_in TEXT,                                -- 最近打卡日期 yyyy-mm-dd
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);
```

## 2. words（词库，含三学段）

```sql
CREATE TABLE words (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  term       TEXT NOT NULL,                -- 英文单词 / 词组
  phonetic   TEXT,                         -- 音标
  meaning    TEXT NOT NULL,               -- 中文释义
  level      INTEGER NOT NULL DEFAULT 1,   -- 高频等级：3=★★★ 2=★★ 1=★
  stage      TEXT NOT NULL,               -- elementary / middle / high
  pos        TEXT,                        -- 词性 n./v./phrase
  usage      TEXT,                        -- 用法说明
  examples   TEXT,                        -- JSON: [{en,zh},...]
  UNIQUE(term, stage)
);
CREATE INDEX idx_words_stage ON words(stage);
```

## 3. word_mastery（单词掌握度，核心追踪表）

```sql
CREATE TABLE word_mastery (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id     INTEGER NOT NULL,
  word_id     INTEGER NOT NULL,
  proficiency INTEGER NOT NULL DEFAULT 0,  -- 0~100 熟练度
  attempts    INTEGER NOT NULL DEFAULT 0,  -- 练习/考试次数
  correct     INTEGER NOT NULL DEFAULT 0,  -- 正确次数
  total_time  INTEGER NOT NULL DEFAULT 0,  -- 累计耗时(ms)
  last_reviewed TEXT,                      -- 最近复习 yyyy-mm-dd
  updated_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(user_id, word_id),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (word_id) REFERENCES words(id) ON DELETE CASCADE
);
-- 学习列表按熟练度从低到高（优先薄弱词）
CREATE INDEX idx_mastery_prof ON word_mastery(user_id, proficiency);
```

## 4. check_ins（打卡记录）

```sql
CREATE TABLE check_ins (
  id        INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id   INTEGER NOT NULL,
  check_date TEXT NOT NULL,                -- yyyy-mm-dd
  UNIQUE(user_id, check_date),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
```

## 5. milestones / user_milestones（里程碑成就）

```sql
CREATE TABLE milestones (
  id      TEXT PRIMARY KEY,               -- m-500-words 等
  title   TEXT NOT NULL,
  desc    TEXT,
  emoji   TEXT,
  reward  INTEGER NOT NULL DEFAULT 0,     -- 解锁奖励积分
  target_kind  TEXT NOT NULL,             -- totalWords/streak/points/mastered
  target_value INTEGER NOT NULL
);

CREATE TABLE user_milestones (
  user_id     INTEGER NOT NULL,
  milestone_id TEXT NOT NULL,
  unlocked_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, milestone_id),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (milestone_id) REFERENCES milestones(id)
);
```

## 6. points_log（积分流水，可选审计）

```sql
CREATE TABLE points_log (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id    INTEGER NOT NULL,
  source     TEXT NOT NULL,               -- check_in / practice / exam / milestone
  delta      INTEGER NOT NULL,            -- 增减积分
  ref_id     INTEGER,                     -- 关联记录 id
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
```

## 7. exam_records（考试明细，用于熟练度计算与回顾）

```sql
CREATE TABLE exam_records (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id    INTEGER NOT NULL,
  word_id    INTEGER NOT NULL,
  q_type     TEXT NOT NULL,               -- zh2en / en2zh
  is_correct INTEGER NOT NULL,
  answer_ms  INTEGER NOT NULL,            -- 单题耗时
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (word_id) REFERENCES words(id) ON DELETE CASCADE
);
```

## 熟练度计算（与 Demo 一致）

```
accuracy   = correct / attempts
avg_time   = total_time / attempts
speedBonus = avg_time <= 6000ms ? +8 : avg_time <= 12000ms ? 0 : -8
proficiency = clamp( 0.5*旧熟练度 + 0.5*(accuracy*100) + speedBonus , 0, 100 )
```
